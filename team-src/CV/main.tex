% Show the bib `nickname` in bold parentheses after the booktitle, e.g. "... Systems (\textbf{NeurIPS})"
\DeclareSourcemap{

    \map{

  \clearname{editor}\clearlist{location}\clearlist{publisher}%
  \clearfield{pages}\clearfield{isbn}}
